=================================================
INSIDE FLINK SQL CLIENT (HIGH SEVERITY METRICS)
=================================================

-- Kafka source table (JSON format)
CREATE TABLE high_severity_stream (
  event_time BIGINT,
  timestamp STRING,
  vibration DOUBLE,
  temperature DOUBLE,
  pressure DOUBLE,
  rms_vibration DOUBLE,
  mean_temp DOUBLE,
  fault_label STRING,
  fuzzy_severity STRING
) WITH (
  'connector' = 'kafka',
  'topic' = 'high_severity',
  'properties.bootstrap.servers' = 'localhost:9092',
  'properties.group.id' = 'flink-high-metrics',
  'scan.startup.mode' = 'earliest-offset',
  'format' = 'json'
);

-- 🔹 AVG LATENCY (ms) – REAL END-TO-END
SELECT
  AVG(UNIX_TIMESTAMP(CURRENT_TIMESTAMP) * 1000 - event_time)
  AS avg_latency_ms
FROM high_severity_stream;

-- 🔹 THROUGHPUT (events / 60 sec)
SELECT
  COUNT(*) AS events_per_min
FROM high_severity_stream
GROUP BY TUMBLE(PROCTIME(), INTERVAL '60' SECOND);
